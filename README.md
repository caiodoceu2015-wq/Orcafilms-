    function verificarSenha() {
        const senhaInserida = document.getElementById('senha').value;
        const senhaCorreta = 'tonito20';

        if (senhaInserida === senhaCorreta) {
            document.getElementById('login-box').style.display = 'none';
            document.getElementById('painel-box').style.display = 'block';
        } else {
            alert('Senha incorreta! Use a senha tonito20 que está na tela.');
        }
    }

    function deslogar() {
        document.getElementById('senha').value = '';
        document.getElementById('login-box').style.display = 'block';
        document.getElementById('painel-box').style.display = 'none';
    }
